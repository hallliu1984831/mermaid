K8S Graceful Shutdown

Pod 退出前先留一点时间，让应用把手里的请求和清理动作处理完。

terminationGracePeriodSeconds: 30
lifecycle:
  preStop:
    exec:
      command: ["sh", "-c", "sleep 10"]
